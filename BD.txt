-- 3. TRIGGER PARA SINCRONIZACIÓN DE USUARIOS

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER SET search_path = public
AS $$
BEGIN
  INSERT INTO public.users (id, email, name, role_id)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'name', 'Usuario Nuevo'),
    1 -- 1 asume que es el ID del rol 'client' en tu tabla roles
  );
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- Permitir al usuario ver solo su propio registro
CREATE POLICY "Usuarios pueden ver su propio perfil" 
ON public.users FOR SELECT 
USING (id = auth.uid());

-- Permitir al usuario actualizar solo sus propios datos (ej. cambiar nombre o teléfono)
CREATE POLICY "Usuarios pueden actualizar su perfil" 
ON public.users FOR UPDATE 
USING (id = auth.uid());

-- TABLA: ADDRESSES
-- Los usuarios tienen control total (ALL: select, insert, update, delete) sobre sus propias direcciones
CREATE POLICY "Usuarios gestionan sus propias direcciones" 
ON public.addresses FOR ALL 
USING (user_id = auth.uid());


-- TABLA: ORDERS
-- 1. Los clientes pueden ver el historial y estado de sus propios pedidos
CREATE POLICY "Clientes ven sus pedidos" 
ON public.orders FOR SELECT 
USING (client_id = auth.uid());

-- 2. Los clientes pueden crear un nuevo pedido (el CHECK asegura que no puedan hacerse pasar por otro)
CREATE POLICY "Clientes crean pedidos" 
ON public.orders FOR INSERT 
WITH CHECK (client_id = auth.uid());

-- 3. Los repartidores pueden ver los pedidos que se les han asignado
CREATE POLICY "Repartidores ven pedidos asignados" 
ON public.orders FOR SELECT 
USING (driver_id = auth.uid());

-- 4. Los repartidores pueden actualizar el estado de su pedido (ej. de 'picked_up' a 'delivered')
CREATE POLICY "Repartidores actualizan estado" 
ON public.orders FOR UPDATE 
USING (driver_id = auth.uid());


-- Permitir que cualquier usuario (incluso los no registrados) vea los restaurantes activos
CREATE POLICY "Todos pueden ver restaurantes activos" 
ON public.restaurants FOR SELECT 
USING (is_active = true);

-- Permitir que todos vean las categorías y el menú
CREATE POLICY "Todos pueden ver categorías de restaurantes" 
ON public.restaurant_categories FOR SELECT USING (true);

CREATE POLICY "Todos pueden ver los items del menú" 
ON public.menu_items FOR SELECT USING (is_active = true);

CREATE POLICY "Todos pueden ver categorías del menú" 
ON public.menu_categories FOR SELECT USING (true);


CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER SET search_path = public
AS $$
BEGIN
  INSERT INTO public.users (id, email, name, phone, role_id)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'name', 'Usuario Nuevo'),
    NEW.phone, -- Aquí capturamos el teléfono que entra por OTP
    1
  );
  RETURN NEW;
END;
$$;
